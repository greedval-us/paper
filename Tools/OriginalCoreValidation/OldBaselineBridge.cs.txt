        static string RunPrepared(string mode,GameBalance balance)
        {
            if(mode!="pressure"&&mode!="campaign")throw new ArgumentException("mode must be pressure or campaign");
            var trace=new TraceEvidence{scenario="original b759182 / actual "+balance.Preset+" asset / "+mode+" / seed1337"};

            if(mode=="pressure")
            {

        static object CopyValue(object source,Type destination,string path)
        {
            if(source==null){if(destination.IsValueType)throw new ArgumentException(path+": null value type");return null;}
            if(destination==typeof(string))return (string)source;
            if(destination.IsEnum)return Enum.ToObject(destination,Convert.ToInt64(source,CultureInfo.InvariantCulture));
            if(destination.IsPrimitive||destination==typeof(decimal))return Convert.ChangeType(source,destination,CultureInfo.InvariantCulture);
            if(destination.IsArray)
            {
                var original=(IEnumerable)source;var items=new List<object>();foreach(var item in original)items.Add(item);
                Type element=destination.GetElementType();var result=Array.CreateInstance(element,items.Count);
                for(int i=0;i<items.Count;i++)result.SetValue(CopyValue(items[i],element,path+"["+i+"]"),i);return result;
            }
            if(destination.IsGenericType&&destination.GetGenericTypeDefinition()==typeof(List<>))
            {
                Type element=destination.GetGenericArguments()[0];var result=(IList)Activator.CreateInstance(destination);int i=0;
                foreach(var item in (IEnumerable)source)result.Add(CopyValue(item,element,path+"["+(i++)+"]"));return result;
            }
            var owned=Activator.CreateInstance(destination);Type actual=source.GetType();
            foreach(var target in destination.GetFields(BindingFlags.Instance|BindingFlags.Public))
            {
                var original=actual.GetField(target.Name,BindingFlags.Instance|BindingFlags.Public);
                if(original==null)throw new ArgumentException(path+"."+target.Name+": source field missing; no default substitution allowed");
                target.SetValue(owned,CopyValue(original.GetValue(source),target.FieldType,path+"."+target.Name));
            }
            return owned;
        }
